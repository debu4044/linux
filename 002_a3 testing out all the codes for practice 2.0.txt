#!/bin/bash

while true
do
    echo "===== FILE OPERATIONS MENU ====="
    echo "1. Create Directory"
    echo "2. Check File Exists"
    echo "3. Create File"
    echo "4. Read File"
    echo "5. Append to File"
    echo "6. Truncate File"
    echo "7. Copy File"
    echo "8. Delete File"
    echo "9. Find File Size"
    echo "10. Check Permissions & File Type"
    echo "11. Exit"

    echo "Enter your choice:"
    read ch

    case $ch in

    1)
        echo "Enter directory name:"
        read dir
        mkdir -p "$dir"
        echo "Directory created"
        ;;

    2)
        echo "Enter file name:"
        read file
        [ -e "$file" ] && echo "Exists" || echo "Does not exist"
        ;;

    3)
        echo "Enter file name:"
        read file
        touch "$file"
        echo "File created"
        ;;

    4)
        echo "Enter file name:"
        read file
        [ -e "$file" ] && cat "$file" || echo "File not found"
        ;;

    5)
        echo "Enter file name:"
        read file
        echo "Enter text:"
        read text
        echo "$text" >> "$file"
        echo "Appended"
        ;;

    6)
        echo "Enter file name:"
        read file
        > "$file"
        echo "File truncated"
        ;;

    7)
        echo "Enter source file:"
        read src
        echo "Enter destination file:"
        read dest
        cp "$src" "$dest"
        echo "File copied"
        ;;

    8)
        echo "Enter file name:"
        read file
        rm -f "$file"
        echo "File deleted"
        ;;

    9)
        echo "Enter file name:"
        read file
        [ -e "$file" ] && ls -l "$file" || echo "File not found"
        ;;

    10)
        echo "Enter file name:"
        read file

        [ -r "$file" ] && echo "Readable"
        [ -w "$file" ] && echo "Writable"
        [ -x "$file" ] && echo "Executable"

        [ -b "$file" ] && echo "Block special file"
        [ -c "$file" ] && echo "Character special file"
        [ -p "$file" ] && echo "Named pipe"

        [ -f "$file" ] && echo "Regular file"
        [ -d "$file" ] && echo "Directory"

        [ -s "$file" ] && echo "Size > 0"

        [ ! -e "$file" ] && echo "File/Directory does not exist"
        ;;

    11)
        echo "Exiting..."
        break
        ;;

    *)
        echo "Invalid choice"
        ;;
    esac

    echo ""
done
