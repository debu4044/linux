Write shell script to find reverse of 5 digit number using while loop

#!/bin/bash
echo "Enter a 5-digit number:"
read num
rev=0
while [ $num -gt 0 ]
do
rem=$((num % 10))
rev=$((rev * 10 + rem))
num=$((num / 10))
done
echo "Reversed number: $rev"

Write script to find sum of cube of 5 digit number 

#!/bin/bash
echo "Enter a 5-digit number:"
read num
sum=0
while [ $num -gt 0 ]
do
    rem=$((num % 10))          # get last digit
    cube=$((rem * rem * rem))  # cube of digit
    sum=$((sum + cube))        # add to sum
    num=$((num / 10))          # remove last digit
done
echo "Sum of cubes of digits: $sum"
